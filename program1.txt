// ---------- CODE: assembler.c ----------

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX 50

// Tables
struct { char name[10]; int addr; } sym[MAX];
struct { char val[10]; int addr; } lit[MAX];
int pool[MAX];

// Counters
int sc = 0, lc = 0, pc = 0, LC = 0;

// Add symbol (no duplicates)
void addSym(char *s) {
    for(int i = 0; i < sc; i++)
        if(strcmp(sym[i].name, s) == 0) return;

    strcpy(sym[sc].name, s);
    sym[sc++].addr = LC;
}

// Add literal
void addLit(char *s) {
    for(int i = 0; i < lc; i++)
        if(strcmp(lit[i].val, s) == 0) return;

    strcpy(lit[lc].val, s);
    lit[lc++].addr = -1;
}

// Assign addresses to literals
void processLit() {
    pool[pc++] = lc;

    for(int i = 0; i < lc; i++)
        if(lit[i].addr == -1)
            lit[i].addr = LC++;
}

int main() {
    FILE *fp = fopen("input.asm", "r");
    char l[100], a[20], b[20], c[20];

    while(fgets(l, 100, fp)) {

        a[0]=b[0]=c[0]='\0';
        sscanf(l, "%s %s %s", a, b, c);

        // START
        if(strcmp(a, "START") == 0) {
            LC = atoi(b);
            continue;
        }

        // LTORG / END
        if(strcmp(a, "LTORG") == 0 || strcmp(a, "END") == 0) {
            processLit();
            continue;
        }

        // Label
        if(a[0] && strcmp(a,"MOVER") && strcmp(a,"ADD"))
            addSym(a);

        // Literal
        if(strstr(c, "='"))
            addLit(c);

        LC++;
    }

    fclose(fp);

    // Output
    printf("\nSYMBOL TABLE:\n");
    for(int i=0;i<sc;i++)
        printf("%s\t%d\n", sym[i].name, sym[i].addr);

    printf("\nLITERAL TABLE:\n");
    for(int i=0;i<lc;i++)
        printf("%s\t%d\n", lit[i].val, lit[i].addr);

    printf("\nPOOL TABLE:\n");
    for(int i=0;i<pc;i++)
        printf("%d\n", pool[i]);

    return 0;
}



// ---------- COMPILE & RUN ----------
