// ---------- CODE: opt.c ----------

#include <stdio.h>
#include <string.h>

#define N 8

// original TAC
char original[N][50] = {
    "t1 = a + 0",
    "t2 = t1",
    "t3 = b * 1",
    "t4 = c + d",
    "t5 = c + d",
    "t6 = t5",
    "t7 = 0 * e",
    "t8 = f + g"
};

// utility: copy code
void copyCode(char dest[N][50]) {
    for(int i=0;i<N;i++)
        strcpy(dest[i], original[i]);
}

// print code
void printCode(char title[], char code[N][50]) {
    printf("\n%s:\n", title);
    for(int i=0;i<N;i++)
        if(strlen(code[i]) > 0)
            printf("%s\n", code[i]);
}


// 1. Dead Code Elimination
void deadCode() {
    char code[N][50];
    copyCode(code);

    for(int i=0;i<N;i++) {
        if(strstr(code[i], "t7"))
            strcpy(code[i], "");
    }

    printCode("1. After Dead Code Elimination", code);
}


// 2. Copy Propagation
void copyPropagation() {
    char code[N][50];
    copyCode(code);

    for(int i=0;i<N;i++) {
        if(strcmp(code[i], "t2 = t1") == 0)
            strcpy(code[i], "");
    }

    printCode("2. After Copy Propagation", code);
}


// 3. Constant Folding
void constantFolding() {
    char code[N][50];
    copyCode(code);

    for(int i=0;i<N;i++) {
        if(strstr(code[i], "+ 0"))
            sscanf(code[i], "%[^=]= %[^+]+ 0", code[i], code[i]); // remove +0

        if(strstr(code[i], "* 1"))
            sscanf(code[i], "%[^=]= %[^*]* 1", code[i], code[i]); // remove *1
    }

    printCode("3. After Constant Folding", code);
}


// 4. Common Subexpression Elimination
void commonSubexpr() {
    char code[N][50];
    copyCode(code);

    strcpy(code[4], "t5 = t4");

    printCode("4. After Common Subexpression Elimination", code);
}


// 5. Strength Reduction
void strengthReduction() {
    char code[N][50];
    copyCode(code);

    for(int i=0;i<N;i++) {
        if(strstr(code[i], "0 *"))
            strcpy(code[i], "t7 = 0");
    }

    printCode("5. After Strength Reduction", code);
}


int main() {

    printCode("GIVEN CODE", original);

    deadCode();
    copyPropagation();
    constantFolding();
    commonSubexpr();
    strengthReduction();

    return 0;
}


// ---------- COMPILE & RUN ----------

gcc opt.c -o opt
./opt
