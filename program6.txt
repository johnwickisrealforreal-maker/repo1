// ---------- CODE: opt.c ----------

#include <stdio.h>
#include <string.h>

#define N 8

// original TAC
char original[N][50] = {
    "t1 = a + 0",
    "t2 = t1",
    "t3 = b * 1",
    "t4 = c + d",
    "t5 = c + d",
    "t6 = t5",
    "t7 = 0 * e",
    "t8 = f + g"
};

// copy utility
void copy(char dest[N][50]) {
    for(int i=0;i<N;i++)
        strcpy(dest[i], original[i]);
}

// print utility
void print(char title[], char code[N][50]) {
    printf("\n%s:\n", title);
    for(int i=0;i<N;i++)
        if(strlen(code[i]) > 0)
            printf("%s\n", code[i]);
}


// 1. Dead Code Elimination
void dead() {
    char code[N][50]; copy(code);

    for(int i=0;i<N;i++)
        if(strstr(code[i], "t7"))
            strcpy(code[i], "");

    print("1. After Dead Code Elimination", code);
}


// 2. Common Subexpression Elimination
void common() {
    char code[N][50]; copy(code);

    strcpy(code[4], "t5 = t4"); // reuse t4

    print("2. After Common Subexpression Elimination", code);
}


// 3. Constant Folding / Simplification
void constant() {
    char code[N][50]; copy(code);

    strcpy(code[0], "t1 = a");   // a + 0 → a
    strcpy(code[2], "t3 = b");   // b * 1 → b
    strcpy(code[6], "t7 = 0");   // 0 * e → 0

    print("3. After Constant Folding", code);
}


int main() {

    print("GIVEN CODE", original);

    dead();
    common();
    constant();

    return 0;
}


// ---------- COMPILE & RUN ----------

gcc opt.c -o opt
./opt
