==============================
PROGRAM 4: YACC - EXPRESSION EVALUATOR (ARITHMETIC + VARIABLES + FUNCTIONS)
==============================


// ---------- INSTALL ----------

sudo apt update
sudo apt install bison flex gcc


// ---------- CREATE FILES ----------

mkdir yacc
cd yacc

touch expr.y expr.l


// ---------- CODE: expr.y ----------

%{
#include <stdio.h>
#include <math.h>
#include <stdlib.h>

int yylex();
void yyerror(char *s);

// simple variable storage
int vars[26];
%}

%union {
    double num;
    int var;
}

%token <num> NUMBER
%token <var> VARIABLE
%type <num> expr
%token SIN COS

%%

input:
    | input line
    ;

line:
    expr '\n'        { printf("Result = %lf\n", $1); }
    | VARIABLE '=' expr '\n' { vars[$1] = $3; }
    ;

expr:
    expr '+' expr    { $$ = $1 + $3; }
    | expr '-' expr  { $$ = $1 - $3; }
    | expr '*' expr  { $$ = $1 * $3; }
    | expr '/' expr  { $$ = $1 / $3; }
    | '(' expr ')'   { $$ = $2; }
    | NUMBER         { $$ = $1; }
    | VARIABLE       { $$ = vars[$1]; }
    | '-' expr       { $$ = -$2; }
    | expr '^' expr  { $$ = pow($1,$3); }
    | SIN '(' expr ')' { $$ = sin($3); }
    | COS '(' expr ')' { $$ = cos($3); }
    ;

%%

void yyerror(char *s) {
    printf("Error: %s\n", s);
}

int main() {
    printf("Enter expressions:\n");
    yyparse();
    return 0;
}


// ---------- CODE: expr.l ----------

%{
#include "y.tab.h"
#include <math.h>
%}

%%

[0-9]+(\.[0-9]+)?   { yylval.num = atof(yytext); return NUMBER; }
[a-z]               { yylval.var = yytext[0]-'a'; return VARIABLE; }

"sin"               { return SIN; }
"cos"               { return COS; }

[\t ]               ;
\n                  return '\n';

.                   return yytext[0];

%%
int yywrap() { return 1; }


// ---------- COMPILE & RUN ----------

yacc -d expr.y
lex expr.l
gcc y.tab.c lex.yy.c -o expr -lm
./expr


// ---------- SAMPLE INPUT ----------

a=5
b=3
a+b
sin(0)
cos(0)
a^b


// ---------- OUTPUT ----------

Result = 8.000000
Result = 0.000000
Result = 1.000000
